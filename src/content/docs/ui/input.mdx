title: Input

The Input components provide a native text field, a group for surrounding content, and a clear button that returns focus to the field.

import { OpenUIRenderReact } from '~/registry/components/openui-preview/openui-render-react';
import { OpenUIRenderVue } from '~/registry/components/openui-preview/openui-render-vue';

export const basicSource = `root = InputGroup({}, { children: [Input({ defaultValue: "hollisgrant.com" }, {}, { placeholder: "example.com", "aria-label": "Website" }), InputClear({}, { children: "Clear" }, { "aria-label": "Clear website" })] }, {})`;
export const searchSource = `root = Input({}, {}, { type: "search", placeholder: "Search people", "aria-label": "Search people" })`;
export const disabledSource = `root = Input({ disabled: true, defaultValue: "hollis@example.com" }, {}, { "aria-label": "Email" })`;
export const readOnlySource = `root = Input({ readOnly: true, value: "example.com/hollis" }, {}, { "aria-label": "Profile URL" })`;
export const groupSource = `root = InputGroup({}, { children: ["https://", Input({}, {}, { placeholder: "example.com", "aria-label": "Website" }), ".com"] }, {})`;
