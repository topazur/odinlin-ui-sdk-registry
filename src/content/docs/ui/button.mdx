---
title: Button
description: Examples for the Odinlin UI SDK button in React and Vue.
---

The Button component runs an action, communicates its state, and stays reachable when disabled. Each example below is rendered through the OpenUI registry in both supported frameworks.

import { OpenUIRenderReact } from '~/registry/components/openui-preview/openui-render-react';
import { OpenUIRenderVue } from '~/registry/components/openui-preview/openui-render-vue';

export const basicSource = `root = Button({}, { children: "Button" })`;
export const disabledSource = `root = Button({ disabled: true }, { children: "Disabled" })`;
export const focusableSource = `root = Button({ disabled: true, focusableWhenDisabled: true }, { children: "Publish" }, { "aria-describedby": "publish-hint" })`;
export const submitSource = `root = Button({ type: "submit" }, { children: "Subscribe" })`;
export const linkSource = `root = Button({}, { children: "Get started" }, { href: "#installation" })`;
export const accessibleSource = `root = Button({}, { children: "Save" }, { accesskey: "s", "aria-label": "Save changes" })`;
